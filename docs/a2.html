// Set the LED and button pin numbers.
const int redLED = 5;
const int greenLED = 4;
const int blueLED = 3;
const int button = 2;

// Track whether the light show is running.
bool running = false;

// Store the previous button reading.
int oldButton = HIGH;

// Run once when the Arduino starts.
void setup() {
// Set LED pins as outputs.
pinMode(redLED, OUTPUT);
pinMode(greenLED, OUTPUT);
pinMode(blueLED, OUTPUT);

// Enable the button's internal pull-up resistor.
pinMode(button, INPUT_PULLUP);
}
// Repeat while the Arduino is powered.
void loop() {
// Read the button's current state.
int newButton = digitalRead(button);

// Detect a new button press.
if (oldButton == HIGH && newButton == LOW) {
// Toggle the light show on or off.
running = !running;

// Reduce repeated button readings.
delay(200);
}

// Save the current button reading.
oldButton = digitalRead(button);

// Run the light show when enabled.
if (running) {
// Turn the red LED on.
digitalWrite(redLED, HIGH);

// Keep the green LED off.
digitalWrite(greenLED, LOW);

// Wait 500 ms and check the button.
if (checkButton(500)) return;

// Turn the red LED off.
digitalWrite(redLED, LOW);

// Turn the green LED on.
digitalWrite(greenLED, HIGH);

// Wait 500 ms and check the button.
if (checkButton(500)) return;

// Turn the green LED off.
digitalWrite(greenLED, LOW);

// Fade the blue LED in gradually.
for (int bright = 0; bright <= 255; bright += 5) {
// Increase the blue LED's brightness.

analogWrite(blueLED, bright);

// Wait 20 ms and check the button.
if (checkButton(20)) return;
}

// Fade the blue LED out gradually.
for (int bright = 255; bright >= 0; bright -= 5) {
// Decrease the blue LED's brightness.
analogWrite(blueLED, bright);

// Wait 20 ms and check the button.

if (checkButton(20)) return;

}

} else {
// Turn off the red LED.
digitalWrite(redLED, LOW);

// Turn off the green LED.

digitalWrite(greenLED, LOW);

// Turn off the blue LED.
analogWrite(blueLED, 0);
}
}


// Check for button presses during a wait.
bool checkButton(int time) {
// Divide the wait into 10 ms intervals.
for (int i = 0; i < time / 10; i++) {

// Wait 10 ms before checking.
delay(10);

// Read the button's current state.
int newButton = digitalRead(button);

// Detect a new button press.

if (oldButton == HIGH && newButton == LOW) {
// Toggle the light show state.
running = !running;

// Save the pressed button state.
oldButton = LOW;

// Reduce repeated button readings.
delay(200);


// Turn off the red LED.

digitalWrite(redLED, LOW);



// Turn off the green LED.
digitalWrite(greenLED, LOW);

// Turn off the blue LED.
analogWrite(blueLED, 0);

// Report that a press was detected.
return true;
}

// Save the latest button reading.
oldButton = newButton;
}

// Report that no press was detected.
return false;
}

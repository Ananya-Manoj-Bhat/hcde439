Schematic for the circuit

This schematic shows my capacitive touch sensor connected to an Arduino Uno and an SG90 servo motor. I used a 1 MΩ resistor between digital pins D4 and D2 to create the capacitive sensing circuit. Two connected coins act as the touch electrode. I chose a 1 MΩ resistor because a high resistance allows the capacitive sensor to detect small changes in capacitance when a person touches the coins. The servo's signal wire is connected to D9, while its power and ground wires are connected to 5V and GND.

set up of the circuit

Here is the code for the circuit


#include
#include

// Touch sensor: send pin 4, receive pin 2
CapacitiveSensor touchSensor = CapacitiveSensor(4, 2);

// Servo connected to pin 9
Servo myServo;

// Touch threshold
const long touchThreshold = 200;

bool wasTouched = false;

void setup() {
Serial.begin(9600);

// Attach servo
myServo.attach(9);
myServo.write(0);

// Disable automatic calibration
touchSensor.set_CS_AutocaL_Millis(0xFFFFFFFF);

Serial.println("Touch sensor ready!");
}

void loop() {
// Read the touch sensor
long sensorValue = touchSensor.capacitiveSensor(30);

// Display sensor reading
Serial.println(sensorValue);

// Detect touch when reading rises above 200
bool isTouched = sensorValue > touchThreshold;

// Move servo when coins are touched
if (isTouched && !wasTouched) {
myServo.write(90);
Serial.println("Touch detected!");
}

// Return servo when coins are released
if (!isTouched && wasTouched) {
myServo.write(0);
Serial.println("Touch released!");
}

wasTouched = isTouched;

delay(50);
}
